COMMANDS

  get FILE PATH          Get value at JSON path
  filter FILE PATH COND  Filter array by condition
  search FILE TERM       Search all keys/values
  keys FILE [PATH]       List keys at path
  stats FILE PATH        Stats on numeric array
  pretty FILE            Pretty-print with color
  validate FILE          Validate JSON syntax
  csv FILE PATH          Convert array to CSV
  count FILE PATH        Count elements

FILTER CONDITIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  key=value      Exact match         name=Alice
  key!=value     Not equal           status!=deleted
  key>num        Greater than        age>18
  key<num        Less than           price<100
  key>=num       Greater or equal    rating>=4.5
  key<=num       Less or equal       score<=10
  key~pattern    Contains            name~ali
  key^pattern    Starts with         email^admin
  key$pattern    Ends with           file$.pdf
  key            Key exists          active
  !key           Key absent/falsy    !deleted

  Booleans: active=true / active=false (lowercase!)

QUICK EXAMPLES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  # Get a value
  jsonquery get data.json "$.name" --raw

  # Get array item by index
  jsonquery get data.json "$.users[0].email" --raw

  # Get all emails
  jsonquery get data.json "$.users[*].email"

  # Filter adults only
  jsonquery filter data.json "$.users" "age>18"

  # Filter active users, show as table
  jsonquery filter data.json "$.users" "active=true" -f table

  # Find by email domain
  jsonquery filter data.json "$.users" "email$example.com"

  # Search everywhere for "admin"
  jsonquery search data.json "admin"

  # Search with regex
  jsonquery search data.json "alice|bob" --regex

  # List keys
  jsonquery keys data.json "$.config"

  # Stats on prices
  jsonquery stats data.json "$.prices"

  # Export to CSV
  jsonquery csv data.json "$.users" > users.csv

  # Validate before deploy
  jsonquery validate config.json

  # Count records
  jsonquery count data.json "$.records"

  # Recursive search all depths
  jsonquery get data.json "$..email"

  # From stdin
  cat data.json | jsonquery get - "$.count"
